cmake_minimum_required(VERSION 3.20)
project(apeq_bench_ole LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(DEFINED BENCH_REVISION AND NOT BENCH_REVISION STREQUAL "")
  set(GIT_SHA "${BENCH_REVISION}")
else()
  set(GIT_SHA "unknown")
endif()
add_definitions(-DGIT_COMMIT="${GIT_SHA}")

set(APEQ_ISA_FLAGS -maes -msse4.1 -msse4.2 -mavx2 -mpclmul)
add_compile_options(${APEQ_ISA_FLAGS} -O3 -Wall -Wextra -Werror)

find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(Boost REQUIRED)
find_package(libOTe CONFIG REQUIRED)

add_executable(driver apeq_eq.cpp)
target_include_directories(driver PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}
  /opt/ips-ole/include
  /opt/ips-ole/adapters)
target_link_libraries(driver PRIVATE
  /opt/ips-ole/build-libote/libips_ole_libote.a
  /opt/ips-ole/build-libote/libips_ole_core.a
  oc::libOTe OpenSSL::Crypto Threads::Threads Boost::headers)
