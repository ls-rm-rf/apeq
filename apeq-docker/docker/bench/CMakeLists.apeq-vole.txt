cmake_minimum_required(VERSION 3.16)
project(apeq_bench_apeq_vole CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(DEFINED BENCH_REVISION AND NOT BENCH_REVISION STREQUAL "")
  set(GIT_SHA "${BENCH_REVISION}")
else()
  set(GIT_SHA "unknown")
endif()
add_definitions(-DGIT_COMMIT="${GIT_SHA}")

set(APEQ_ISA_FLAGS -maes -msse4.1 -msse4.2 -mavx2 -mpclmul
    CACHE STRING "ISA flags shared by all benchmark targets")
add_compile_options(${APEQ_ISA_FLAGS} -O3 -Wno-deprecated-declarations)

find_package(volePSI REQUIRED)
include_directories(${CMAKE_CURRENT_SOURCE_DIR})

add_executable(driver apeq_vole_eq.cpp)
target_link_libraries(driver visa::volePSI)
